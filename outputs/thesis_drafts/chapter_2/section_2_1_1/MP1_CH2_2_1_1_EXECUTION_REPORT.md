## Công thức tái dựng Word

Lưu block sau ra scratch dưới tên rebuild_word.py. Chạy với CSV truy vết và đích DOCX mới; script từ chối ghi đè. Không cần nguồn article để dựng lại nội dung Word; muốn rà khoa học phải mở nguồn và hash như manifest.

```python
from pathlib import Path
import csv, io, zipfile
from datetime import datetime, timezone
from docx import Document
from docx.shared import Cm, Pt, RGBColor
from docx.enum.text import WD_ALIGN_PARAGRAPH
from docx.oxml import OxmlElement
from docx.oxml.ns import qn

def build(trace_path, output_path):
    rows = list(csv.DictReader(Path(trace_path).open(encoding='utf-8-sig')))
    paragraphs = {}
    references = {}
    for row in rows:
        pid = row['paragraph_id']
        if pid in paragraphs:
            assert paragraphs[pid] == row['manuscript_paragraph_text']
        paragraphs[pid] = row['manuscript_paragraph_text']
        if row['citation_number']:
            references[int(row['citation_number'])] = row['bibliographic_identity']
    doc = Document()
    section = doc.sections[0]
    section.page_width, section.page_height = Cm(21), Cm(29.7)
    section.left_margin, section.right_margin = Cm(3), Cm(2)
    section.top_margin, section.bottom_margin = Cm(2.5), Cm(2.5)
    for style in doc.styles:
        if style.type == 1 or style.type == 2:
            style.font.name = 'Times New Roman'
            style.font.size = Pt(12)
            style.font.color.rgb = RGBColor(0, 0, 0)
            rpr = style.element.get_or_add_rPr()
            fonts = rpr.find(qn('w:rFonts'))
            if fonts is None:
                fonts = OxmlElement('w:rFonts'); rpr.append(fonts)
            for key in list(fonts.attrib):
                if 'theme' in key.lower():
                    del fonts.attrib[key]
            for key in ['ascii', 'hAnsi', 'eastAsia', 'cs']:
                fonts.set(qn('w:' + key), 'Times New Roman')
            lang = OxmlElement('w:lang'); lang.set(qn('w:val'), 'vi-VN'); rpr.append(lang)
    normal = doc.styles['Normal'].paragraph_format
    normal.line_spacing = 1.15
    normal.space_after = Pt(6)
    normal.space_before = Pt(0)
    normal.first_line_indent = Cm(.75)
    normal.widow_control = True
    for name in ['Heading 2', 'Heading 3']:
        style = doc.styles[name]
        style.font.bold = True
        style.paragraph_format.first_line_indent = Cm(0)
        style.paragraph_format.space_before = Pt(0)
        style.paragraph_format.space_after = Pt(8)
        style.paragraph_format.line_spacing = 1.15
        style.paragraph_format.keep_with_next = True
    doc.add_heading('2.1.1. Mục đích của chương tổng quan', level=3)
    for pid in sorted(paragraphs):
        p = doc.add_paragraph(paragraphs[pid])
        p.alignment = WD_ALIGN_PARAGRAPH.JUSTIFY
    doc.add_heading('Tài liệu tham khảo', level=2)
    for number in sorted(references):
        p = doc.add_paragraph(f'[{number}] {references[number]}')
        p.paragraph_format.left_indent = Cm(.75)
        p.paragraph_format.first_line_indent = Cm(-.75)
        p.paragraph_format.space_after = Pt(4)
    cp = doc.core_properties
    cp.title = '2.1.1. Mục đích của chương tổng quan'
    cp.subject = 'MP1; bản thảo tiểu mục; chờ nghiệm thu độc lập'
    cp.author = 'MP1'
    cp.last_modified_by = 'MP1'
    cp.created = cp.modified = datetime(2026, 10, 20, tzinfo=timezone.utc)
    cp.revision = 1
    cp.comments = 'SELF_REVIEW_ONLY; HUMAN_APPROVAL_PENDING'
    buffer = io.BytesIO(); doc.save(buffer)
    with zipfile.ZipFile(buffer) as source, zipfile.ZipFile(output_path, 'w', zipfile.ZIP_DEFLATED) as target:
        for name in sorted(source.namelist()):
            info = zipfile.ZipInfo(name, (2026, 10, 20, 0, 0, 0))
            info.compress_type = zipfile.ZIP_DEFLATED
            target.writestr(info, source.read(name))

if __name__ == '__main__':
    import sys
    assert len(sys.argv) == 3, 'Usage: rebuild_word.py TRACE.csv OUTPUT.docx'
    assert not Path(sys.argv[2]).exists(), 'Refuse to overwrite existing output'
    build(sys.argv[1], sys.argv[2])
```
